import sys
import rclpy
from rclpy.node import Node
from rclpy.exceptions import ParameterNotDeclaredException
from sys import exit


class CameraReconfigure(Node):
    def __init__(self):
        super().__init__('camera_reconfigure')
        self.declare_parameter('exposure_auto', 1)
        self.declare_parameter('white_balance_auto', 1)

    def enable_auto(self):
        self.set_parameters([
            rclpy.parameter.Parameter('exposure_auto', rclpy.Parameter.Type.INTEGER, 1),
            rclpy.parameter.Parameter('white_balance_auto', rclpy.Parameter.Type.INTEGER, 1)
        ])
        self.get_logger().info('Enabled automatic exposure and white balance')

    def disable_auto(self):
        self.set_parameters([
            rclpy.parameter.Parameter('exposure_auto', rclpy.Parameter.Type.INTEGER, 0),
            rclpy.parameter.Parameter('white_balance_auto', rclpy.Parameter.Type.INTEGER, 0)
        ])
        self.get_logger().info('Disabled automatic exposure and white balance')


if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: camera_reconfigure --enable/disable")
        exit(-1)

    rclpy.init()
    reconfigure = CameraReconfigure()

    if sys.argv[1] == "--enable":
        reconfigure.enable_auto()
    else:
        reconfigure.disable_auto()

    rclpy.sleep(1)
    reconfigure.destroy_node()
    rclpy.shutdown()
