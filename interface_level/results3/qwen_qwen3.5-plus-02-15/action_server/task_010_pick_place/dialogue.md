void pick(moveit::planning_interface::MoveGroupInterface& move_group)
{
  std::vector<moveit_msgs::msg::PlaceLocation> place_locations;
  
  // Generate pick locations (simplified for this example, usually involves grasp generation)
  // Here we just define a grasp pose relative to the object
  
  moveit_msgs::msg::Grasp grasp;
  
  // Set the gripper approach
  geometry_msgs::msg::PoseStamped approach_pose;
  approach_pose.header.frame_id = "panda_link0";
  approach_pose.pose.position.x = 0.4;
  approach_pose.pose.position.y = 0.0;
  approach_pose.pose.position.z = 0.5;
  approach_pose.pose.orientation.w = 1.0;
  
  grasp.grasp_pose = approach_pose;
  
  // Pre-grasp approach (move straight down)
  grasp.pre_grasp_approach.direction.header.frame_id = "panda_link0";
  grasp.pre_grasp_approach.direction.vector.z = -1.0;
  grasp.pre_grasp_approach.min_distance = 0.095;
  grasp.pre_grasp_approach.desired_distance = 0.115;
  
  // Post-grasp retreat (move up)
  grasp.post_grasp_retreat.direction.header.frame_id = "panda_link0";
  grasp.post_grasp_retreat.direction.vector.z = 1.0;
  grasp.post_grasp_retreat.min_distance = 0.1;
  grasp.post_grasp_retreat.desired_distance = 0.25;
  
  // Set the gripper posture for grasping (closed)
  closedGripper(grasp.grasp_posture);
  
  // Set the gripper posture for pre-grasp (open)
  openGripper(grasp.pre_grasp_posture);
  
  // Set support surface (table1)
  move_group.setSupportSurfaceName("table1");
  
  // Call pick with the object id and the grasp
  // Note: In a real scenario, you would pass a vector of grasps. 
  // For this migration, we pass a vector containing our single grasp.
  std::vector<moveit_msgs::msg::Grasp> grasps;
  grasps.push_back(grasp);
  
  move_group.pick("object", grasps);
}

void place(moveit::planning_interface::MoveGroupInterface& group)
{
  std::vector<moveit_msgs::msg::PlaceLocation> place_locations;
  
  moveit_msgs::msg::PlaceLocation place_location;
  
  // Set the place pose (on table2)
  geometry_msgs::msg::PoseStamped place_pose;
  place_pose.header.frame_id = "panda_link0";
  place_pose.pose.position.x = 0.0;
  place_pose.pose.position.y = 0.5;
  place_pose.pose.position.z = 0.5;
  place_pose.pose.orientation.w = 1.0;
  
  place_location.place_pose = place_pose;
  
  // Pre-place approach (move up before placing)
  place_location.pre_place_approach.direction.header.frame_id = "panda_link0";
  place_location.pre_place_approach.direction.vector.z = 1.0;
  place_location.pre_place_approach.min_distance = 0.1;
  place_location.pre_place_approach.desired_distance = 0.2;
  
  // Post-place retreat (move away after placing)
  place_location.post_place_retreat.direction.header.frame_id = "panda_link0";
  place_location.post_place_retreat.direction.vector.x = -1.0;
  place_location.post_place_retreat.min_distance = 0.1;
  place_location.post_place_retreat.desired_distance = 0.25;
  
  // Set gripper posture to open for releasing
  openGripper(place_location.post_place_posture);
  
  place_locations.push_back(place_location);
  
  // Set support surface (table2)
  group.setSupportSurfaceName("table2");
  
  // Call place
  group.place("object", place_locations);
}
